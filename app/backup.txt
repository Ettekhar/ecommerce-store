// app/layout.tsx
import { useEffect, useState } from 'react';
import axios from 'axios';
import { Toaster, toast } from 'react-hot-toast';
import type { Metadata } from 'next';
import { Urbanist } from 'next/font/google';

import Footer from '@/components/footer';
import Navbar from '@/components/navbar';
import ModalProvider from '@/providers/modal-provider';
import { ToasterProvider } from '@/providers/toast-provider';

import './globals.css';

const font = Urbanist({ subsets: ['latin'] });

export const metadata: Metadata = {
  title: 'Store',
  description: 'Store',
};

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  const [isBlocked, setIsBlocked] = useState<boolean>(false);

  useEffect(() => {
    const fetchIpStatus = async () => {
      try {
        const ipResponse = await axios.get(`${process.env.NEXT_PUBLIC_API_URL}/get-user-ip`);
        const ipAddress = ipResponse.data.ip;

        const response = await axios.get(`${process.env.NEXT_PUBLIC_API_URL}/block-ip/${ipAddress}`);
        if (response.data.exists) {
          setIsBlocked(true);
          toast.error("Your IP address is blocked for 3 days.");
        }
      } catch (error) {
        console.error('Error fetching IP status:', error);
      }
    };

    fetchIpStatus();
  }, []);

  return (
    <html lang="en">
      <body className={font.className}>
        <ModalProvider />
        <ToasterProvider />
        <Navbar />
        {children}
        <Footer />
        <Toaster />
      </body>
    </html>
  );
}
